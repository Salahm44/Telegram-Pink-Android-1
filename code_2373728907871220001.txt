  async hashSecret(value, salt) {
    const data = new TextEncoder().encode(String(salt) + ":" + String(value));
    const hash = await crypto.subtle.digest("SHA-256", data);
    return Array.from(new Uint8Array(hash))
        .map(b => b.toString(16).padStart(2, "0"))
        .join("");
  }

  esc(value) {
    return String(value ?? "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
  }

  safeMediaSrc(src) {
    if (!src) return "";
    const value = String(src).trim();
    if (
        value.startsWith("blob:") ||
        value.startsWith("https://") ||
        value.startsWith("data:image/") ||
        value.startsWith("data:audio/") ||
        value.startsWith("data:video/") ||
        value.startsWith("data:application/")
    ) {
        return value;
    }
    return "";
  }

  playNotes(notes) {
    try {
      const ctx = this.getAudioCtx();
      if (!ctx) return;
      if (ctx.state === 'suspended') ctx.resume().catch(() => {});
      const t0 = ctx.currentTime + 0.01;
      notes.forEach(([f, start, dur, wave, vol, endF]) => {
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = wave || 'sine';
        osc.frequency.setValueAtTime(f, t0 + start);
        if (endF) osc.frequency.exponentialRampToValueAtTime(endF, t0 + start + dur);
        gain.gain.setValueAtTime(0.0001, t0 + start);
        gain.gain.exponentialRampToValueAtTime(vol || 0.2, t0 + start + 0.012);
        gain.gain.exponentialRampToValueAtTime(0.0001, t0 + start + dur);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start(t0 + start);
        osc.stop(t0 + start + dur + 0.03);
      });
    } catch (e) {}
  }

  getDataUrlByteSize(dataUrl) {
    try {
      const comma = String(dataUrl || '').indexOf(',');
      if (comma < 0) return 0;
      const body = String(dataUrl).slice(comma + 1);
      if (/;base64/i.test(String(dataUrl).slice(0, comma))) return Math.floor(body.length * 0.75);
      return decodeURIComponent(body).length;
    } catch(e) { return 0; }
  }

  async uploadMediaForMessage(dataUrl, mediaType) {
    let blob = dataUrl instanceof Blob ? dataUrl : null;
    if(!blob){
      const safe=this.safeMediaSrc(dataUrl);
      if(!safe || !safe.startsWith('data:')) return safe;
      const comma=safe.indexOf(',');
      if(comma<0) throw new Error('INVALID_MEDIA');
      const header=safe.slice(0,comma), body=safe.slice(comma+1);
      const mime=(header.match(/^data:([^;]+)/i)||[,'application/octet-stream'])[1];
      if(/;base64/i.test(header)){const binary=atob(body),bytes=new Uint8Array(binary.length);for(let i=0;i<binary.length;i++)bytes[i]=binary.charCodeAt(i);blob=new Blob([bytes],{type:mime});}
      else blob=new Blob([decodeURIComponent(body)],{type:mime});
    }
    if(blob.size>10*1024*1024) throw new Error('FILE_TOO_LARGE');
    const mime=blob.type || (mediaType==='video'?'video/mp4':'audio/mpeg');
    const ext=mediaType==='video'?'mp4':(mime.split('/')[1]||'bin').replace(/[^a-z0-9]/gi,'').slice(0,8)||'bin';
    const path=`chatMedia/${String(this.currentUser.phone).replace(/[^0-9A-Za-z_-]/g,'_')}/${Date.now()}_${Math.random().toString(36).slice(2,9)}.${ext}`;
    const uploadRef=storageRef(storage,path);
    await uploadBytes(uploadRef,blob,{contentType:mime,cacheControl:'public,max-age=31536000'});
    return await getDownloadURL(uploadRef);
  }

  async startDirectAudioPlayback(msgId, forceManual=false) {
    const msg = (this.currentList || []).find(x => String(x.id) === String(msgId));
    if (!msg || msg.mediaType !== 'audio' || !msg.directAudio) return false;
    const key = this.getAutoAudioPlayedKey(msgId);
    if ((!forceManual && localStorage.getItem(key) === '1') || this.autoAudioStarting.has(String(msgId))) return false;
    const src = this.loadedMediaUrls[msgId];
    if (!src) return false;
    const box = document.getElementById('messages');
    const audio = box?.querySelector(`audio[data-message-id="${CSS.escape(String(msgId))}"]`);
    if (!audio) return false;
    this.autoAudioStarting.add(String(msgId));
    try {
      const result = audio.play();
      if (result && typeof result.then === 'function') await result;
      // يسجل التشغيل بعد نجاحه حتى لا يعاد تلقائياً عند إغلاق المحادثة وفتحها.
      if (!forceManual) localStorage.setItem(key, '1');
      return true;
    } catch (e) {
      console.log('Direct audio autoplay was blocked:', e);
      return false;
    } finally {
      this.autoAudioStarting.delete(String(msgId));
    }
  }

  }

  async authenticateWithCredentials(phone, pin, silent = false) {
    const status = document.getElementById("authStatus");
    try {
      await ensureAuth();
      const cleanPhone = String(phone).replace(/\D/g, '');
      if (!/^\d{4,20}$/.test(pin)) {
        if (!silent) status.textContent = "رمز PIN غير صالح.";
        return;
      }
      const snap = await get(ref(db, `users/${cleanPhone}`));
      if (snap.exists()) {
        const userData = snap.val();
        if (userData.banned) {
          if (!silent) status.textContent = "هذا الحساب محظور من دخول التطبيق.";
          return;
        }

        let verified = false;
        let pinHash = "";
        let pinSalt = "";

        if (userData.pin && !userData.pinHash) {
          if (userData.pin === pin) {
            pinSalt = crypto.randomUUID();
            pinHash = await this.hashSecret(pin, pinSalt);
            await update(ref(db, `users/${cleanPhone}`), {
              pinHash: pinHash,
              pinSalt: pinSalt,
              pin: null
            });
            verified = true;
          }
        } else if (userData.pinHash && userData.pinSalt) {
          pinSalt = userData.pinSalt;
          pinHash = await this.hashSecret(pin, pinSalt);
          if (pinHash === userData.pinHash) {
            verified = true;
          }
        }

        if (verified) {
          const isAdminCheck = (cleanPhone === "734571128");
          if (!userData.uid) {
            userData.uid = auth.currentUser.uid;
            await update(ref(db, `users/${cleanPhone}`), { uid: auth.currentUser.uid });
          }
          this.currentUser = { ...userData, pinHash, pinSalt };
          delete this.currentUser.pin;
          this.cacheProfile();
          this.isAdmin = isAdminCheck;
          this.updateBotActivationLinks(cleanPhone);
          const saveSessionChoice = document.getElementById("sessionSaveSetting").value;
          if (saveSessionChoice === 'save') {
            localStorage.setItem("sms_user_phone", cleanPhone);
            localStorage.setItem("sms_user_pin_hash", pinHash);
            localStorage.setItem("sms_save_session", "save");
          } else {
            localStorage.removeItem("sms_user_phone");
            localStorage.removeItem("sms_user_pin_hash");
            localStorage.setItem("sms_save_session", "nosave");
          }
          this.setupPresence();
          this.initUserBanListener();
          this.showAppScreen();
        } else {
          if (!silent) status.textContent = "رمز PIN غير صحيح!";
        }
      } else {
        if (!silent) {
          window.tempPhone = cleanPhone;
          document.getElementById("setupPin").value = pin;
          document.getElementById("authScreen").classList.add("hidden");
          document.getElementById("profileScreen").classList.remove("hidden");
        }
      }
    } catch (e) {
      if (!silent) status.textContent = "فشل الاتصال بقاعدة بيانات Firebase.";
    }
  }

  async authenticateWithHash(phone, pinHash, silent = false) {
    const status = document.getElementById("authStatus");
    const cleanPhone = String(phone).replace(/\D/g, '');
    let cached = null;
    try { cached = JSON.parse(localStorage.getItem('sms_profile_cache') || 'null'); } catch (e) {}

    // دخول محلي كامل عند انقطاع الإنترنت: الحساب المحفوظ + PIN المحفوظ
    // يكفيان لفتح التطبيق وقراءة الرسائل المخزنة محلياً.
    if (cached && String(cached.phone) === cleanPhone && !cached.banned) {
      this.currentUser = { ...cached, pinHash };
      this.isAdmin = (cleanPhone === "734571128");
      this.updateBotActivationLinks(cleanPhone);
      this.applyTheme();
      this.showAppScreen();
      if (navigator.onLine !== false) {
        try {
          await ensureAuth();
          get(ref(db, `users/${cleanPhone}`)).then(snap => {
            const u = snap.val();
            if (!u || u.banned || u.pinHash !== pinHash) return;
            this.currentUser = { ...u, pinHash };
            delete this.currentUser.pin;
            this.cacheProfile();
          }).catch(() => {});
        } catch (e) {}
      }
      return;
    }

    try {
      await ensureAuth();

      const snap = await get(ref(db, `users/${cleanPhone}`));
      if (snap.exists()) {
        const userData = snap.val();
        if (userData.banned || userData.pinHash !== pinHash) {
          if (!silent) status.textContent = "جلسة غير صالحة أو الحساب محظور.";
          return;
        }
        this.currentUser = { ...userData };
        delete this.currentUser.pin;
        this.isAdmin = (cleanPhone === "734571128");
        this.cacheProfile();
        this.updateBotActivationLinks(cleanPhone);
        this.setupPresence();
        this.initUserBanListener();
        this.showAppScreen();
      }
    } catch(e) {}
  }

  async setupPeerConnection(roomId) {
    if (this.peerConnection) return this.peerConnection;
    const configuration = { iceServers: [{ urls: 'stun:stun.l.google.com:19302' }] };
    this.peerConnection = new RTCPeerConnection(configuration);

    this.localStream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false });
    this.localStream.getTracks().forEach(track => {
      this.peerConnection.addTrack(track, this.localStream);
    });

    this.peerConnection.ontrack = event => {
      const remoteAudio = document.getElementById("remoteAudio");
      remoteAudio.srcObject = event.streams[0];
      remoteAudio.play().catch(e => console.log("Audio play error", e));
    };

    this.peerConnection.onicecandidate = event => {
      if (event.candidate) {
        const isCaller = this.callData && this.callData.caller === this.currentUser.phone;
        const candidatePath = isCaller ? `calls/${roomId}/callerCandidates` : `calls/${roomId}/calleeCandidates`;
        push(ref(db, candidatePath), event.candidate.toJSON());
      }
    };

    return this.peerConnection;
  }

  async startVoiceCall() {
    if (!this.chatRoomId || !this.activeFriendPhone) return;
    if (this.blockedPhones.has(this.activeFriendPhone)) {
      alert("هذا المستخدم محظور.");
      return;
    }
    const roomId = this.chatRoomId;
    const callModal = document.getElementById("callModal");
    const statusText = document.getElementById("callStatusText");
    const incomingActions = document.getElementById("incomingCallActions");

    callModal.classList.remove("hidden");
    incomingActions.classList.add("hidden");
    statusText.textContent = "جاري الاتصال الصوتي...";

    try {
      const pc = await this.setupPeerConnection(roomId);
      const offer = await pc.createOffer();
      await pc.setLocalDescription(offer);

      const callRef = ref(db, `calls/${roomId}`);
      this.callData = {
        caller: this.currentUser.phone,
        callee: this.activeFriendPhone,
        status: 'ringing',
        offer: { type: offer.type, sdp: offer.sdp }
      };
      await set(callRef, this.callData);
      this.startCallRingtone('ringback');

      this.callUnsubscribe = onValue(callRef, async snapshot => {
        const data = snapshot.val();
        if (!data) {
          this.endVoiceCall(false);
          return;
        }
        if (data.status === 'answered' && pc.remoteDescription == null && data.answer) {
          this.stopCallRingtone();
          statusText.textContent = "المكالمة جارية...";
          const remoteDesc = new RTCSessionDescription(data.answer);
          await pc.setRemoteDescription(remoteDesc);
        } else if (data.status === 'rejected' || data.status === 'ended') {
          this.stopCallRingtone();
          this.endVoiceCall(false);
        }
      });

      const calleeCandidatesRef = ref(db, `calls/${roomId}/calleeCandidates`);
      this.iceUnsubscribe = onChildAdded(calleeCandidatesRef, snapshot => {
        const candidate = new RTCIceCandidate(snapshot.val());
        if (this.peerConnection) {
          this.peerConnection.addIceCandidate(candidate).catch(e => console.log(e));
        }
      });

    } catch (e) {
      alert("تعذر الوصول للميكروفون أو بدء المكالمة.");
      this.endVoiceCall();
    }
  }

  listenForIncomingCalls(roomId) {
    if (this.callUnsubscribe) return;
    const callRef = ref(db, `calls/${roomId}`);
    this.callUnsubscribe = onValue(callRef, async snapshot => {
      const data = snapshot.val();
      if (!data) return;

      if (this.blockedPhones.has(data.caller)) return;

      if (data.status === 'ringing' && data.caller !== this.currentUser.phone) {
        this.callData = data;
        const callModal = document.getElementById("callModal");
        const statusText = document.getElementById("callStatusText");
        const incomingActions = document.getElementById("incomingCallActions");

        callModal.classList.remove("hidden");
        incomingActions.classList.remove("hidden");
        statusText.textContent = `مكالمة صوتية واردة من ${data.caller}...`;
        
        this.startCallRingtone();
      } else if (data.status === 'ended' || data.status === 'rejected') {
        this.stopCallRingtone();
        this.endVoiceCall(false);
      }
    });
  }

  loadContacts() {
    if (this.contactsUnsubscribe) { this.renderContactsList(); return; }

    importOverlay.classList.add("hidden");

    if (searchInput && !searchInput.oninput) {
      searchInput.oninput = () => this.renderContactsList();
    }

    const me = String(this.currentUser.phone);
    const cacheKey = `sms_contacts_cache_${me}`;
    try {
      const c = JSON.parse(localStorage.getItem(cacheKey) || 'null');
      if (c) { this.latestContacts = c; this.rebuildBlocked(); this.renderContactsList(); }
    } catch (e) {}

  rebuildBlocked() {

      if (c && c.blocked) {
        this.blockedPhones.add(phoneKey);
        if (c.phone) this.blockedPhones.add(c.phone);
      }
    });
  }

  syncPresenceWatchers() {
    if (!this.currentUser) return;
    const want = new Set();
    Object.values(this.latestContacts || {}).forEach(c => { if (c && c.phone && !c.blocked) want.add(String(c.phone)); });
    Object.keys(this.inboxData || {}).forEach(p => want.add(String(p)));

    want.forEach(p => {
      if (this.presenceWatchers[p]) return;
      const info = this.userInfo[p] = this.userInfo[p] || {};
      this.presenceWatchers[p] = onValue(ref(db, `users/${p}/online`), (s) => {
        const now = s.val() === true;
        if (info.online !== now) { info.online = now; this.scheduleRender(); }
      }, () => {});
      if (!info._fetched) {
        info._fetched = true;
        const isContact = !!(this.latestContacts || {})[p];
        const tasks = isContact
          ? [get(ref(db, `users/${p}/telegramChatId`))]
          : [get(ref(db, `users/${p}/name`)), get(ref(db, `users/${p}/customId`)), get(ref(db, `users/${p}/telegramChatId`))];
        Promise.all(tasks).then(r => {
          if (isContact) { info.telegramChatId = r[0].val() || ''; }
          else { info.name = r[0].val() || ''; info.customId = r[1].val() || ''; info.telegramChatId = r[2].val() || ''; }
          this.scheduleRender();
        }).catch(() => {});
      }
    });
    Object.keys(this.presenceWatchers).forEach(p => {
      if (!want.has(p)) { try { this.presenceWatchers[p](); } catch (e) {} delete this.presenceWatchers[p]; }
    });
  }

  unreadBadgeHtml(n) {
    if (!n || n < 1) return '';
    const label = n === 2 ? 'رسالتان جديدتان' : (n >= 3 && n <= 10 ? 'رسائل جديدة' : 'رسالة جديدة');
    const num = n > 99 ? '99+' : n;
    return `<div style="margin-bottom: 4px;"><span class="unread-badge">📬 ${num} ${label}</span></div>`;
  }

  scheduleRender() {

    this._renderTimer = setTimeout(() => this.renderContactsList(), 120);

  syncRoomWatchers() {
    // لم يعد التطبيق يفتح مستمعين لكل غرف المحادثات.
    // الرسائل الواردة تصل من صندوق المستخدم فقط، أما الرسائل القديمة فتؤخذ من التخزين المحلي.
    return;
  }

  attachIncomingInboxListener(me) {
    const inboxRef = ref(db, `inbox/${me}`);
    const processed = new Set();

    const processIncoming = async (snap) => {
      try {
        const inboxId = snap.key;
        const m = snap.val();
        if (!inboxId || !m || m.isCommand) return;

        const sender = String(m.senderId || '').trim();
        const recipient = String(m.recipientId || '').trim();
        if (!sender || recipient !== me) return;
        if (sender === me) return;

        // إذا كانت الرسالة قد استلمت سابقاً، فلا تعالج مرة أخرى.
        // يمكن أن يعيد Firebase إرسال child عند إعادة فتح التطبيق؛ receivedAt يمنع التكرار.
        if (m.receivedAt || m.status === 'read' || processed.has(inboxId)) return;

        const roomId = String(m.roomId || (m.isBroadcast ? `broadcast_${me}` : [me, sender].sort().join('_')));
        if (!this.roomsData[roomId]) this.roomsData[roomId] = {};

        // تحقق إضافي من التخزين المحلي: إذا كانت الرسالة محفوظة بالفعل، لا نعيد إضافتها.
        let alreadyLocal = false;
        try {
          const cached = JSON.parse(localStorage.getItem(`sms_local_msgs_${roomId}`) || '[]');
          alreadyLocal = Array.isArray(cached) && cached.some(x => String(x.id) === String(inboxId));
        } catch (e) {}

        const localMessage = {
          ...m,
          id: inboxId,
          // الاستلام لا يعني القراءة: يبقى unread حتى يفتح المستخدم المحادثة.
          status: m.status || 'delivered',
          receivedAt: this.nowServer()
        };

        // بيانات مختصرة للقائمة؛ لا نحفظ الرسالة في inboxData كمرجع تاريخي.
        this.inboxData[sender] = {
          lastAt: Number(m.createdAt || Date.now()),
          messageId: inboxId
        };
        this.roomsData[roomId][inboxId] = alreadyLocal ? {
          ...(this.roomsData[roomId][inboxId] || localMessage),
          ...localMessage
        } : localMessage;

        // حفظ فوري على الهاتف قبل أي حذف من Firebase.
        this.saveLocalRoomCache(roomId);
        processed.add(inboxId);

        // تثبيت حالة الاستلام على عنصر inbox نفسه؛ هذا هو القفل الذي يمنع
        // إعادة استلام الرسالة عند إعادة تشغيل التطبيق قبل مرور 90 ثانية.
        const receivedAt = Number(localMessage.receivedAt) || Date.now();
        update(ref(db, `inbox/${me}/${inboxId}`), {
          status: 'read',
          receivedAt: serverTimestamp(),
          roomId: roomId
        }).catch(() => {});

        // نحدّث نسخة الغرفة أيضاً حتى تبقى متوافقة مع التخزين المحلي.
        update(ref(db, `chatRooms/${roomId}/messages/${inboxId}`), {
          status: 'read',
          receivedAt: serverTimestamp()
        }).catch(() => {});

        if (this.chatRoomId === roomId) {
          const friendName = this.currentFriendName || sender;
          const list = Object.entries(this.roomsData[roomId] || {})
            .map(([id, x]) => ({ id, ...(x || {}) }))
            .filter(x => !x.isCommand && !this.isDeletedForMe(x) && !this.blockedPhones.has(x.senderId))
            .sort((a,b) => Number(a.createdAt||0) - Number(b.createdAt||0))
            .slice(-300);
          await this.renderMessagesList(list, friendName);
          this.autoPlayLatestDirectAudio(list).catch(() => {});
        } else {
          this.notifyIncoming(roomId, localMessage);
        }

        this.scheduleRender();

        // يبدأ عداد 90 ثانية بعد الاستلام الفعلي، وليس بعد الإرسال.
        setTimeout(() => {
          this.removeServerMessage(roomId, inboxId, me);
        }, RECEIVED_TTL_MS);
      } catch (e) {
        console.error('incoming inbox message:', e);
      }
    };

    const uAdd = onChildAdded(inboxRef, processIncoming, (err) => {
      console.error('inbox listener:', err);
    });

    return () => { try { uAdd(); } catch (e) {} };
  }

  attachRoomWatcher(rId) {
    // متوافق مع الاستدعاءات القديمة داخل الكود: لا يتم الاتصال بخادم chatRooms هنا.
    if (!this.roomsData[rId]) this.roomsData[rId] = {};
    this.loadLocalRoomCache(rId);
    if (!this.roomWatchers[rId]) this.roomWatchers[rId] = () => {};
  }

  detachAllRoomWatchers() {
    Object.values(this.roomWatchers).forEach(u => { try { u(); } catch (e) {} });
    this.roomWatchers = {};
    this.roomsData = {};
    this.knownMsgIds = {};
  }

  notifyIncoming(rId, m) {
    if (this.chatRoomId === rId) return;
    const sender = String(m.senderId || '').trim();
    if (this.blockedPhones.has(sender)) return;
    const c = (this.latestContacts || {})[sender];
    if (c && c.muted) return;
    const _t = Date.now();
    if (_t - (this._lastTone || 0) < 1000) return;
    this._lastTone = _t;
    this.playTone('incomingMsg');
  }

  startSweeper() {
    if (this.sweepTimer) return;
    this.sweepTimer = setInterval(() => this.sweepReceivedMessages(), 30000);
    this.sweepReceivedMessages();
  }

  sweepReceivedMessages() {
    if (!this.currentUser) return;
    const now = this.nowServer();

    // تنظيف دوري احتياطي: أي رسالة سبق أن سجل التطبيق استلامها
    // ولم تُحذف بسبب انقطاع مؤقت، تُحذف بعد 90 ثانية من receivedAt.
    for (const [rId, msgs] of Object.entries(this.roomsData || {})) {
      for (const [mId, m] of Object.entries(msgs || {})) {
        if (!m) continue;
        const received = Number(m.receivedAt) || 0;
        if (received && (now - received >= RECEIVED_TTL_MS)) {
          this.removeServerMessage(rId, mId);
        }
      }
    }
  }

  removeServerMessage(rId, mId, recipientPhone = '') {
    const key = `${rId}/${mId}`;
    if (this.removing.has(key)) return;
    this.removing.add(key);

    const jobs = [
      remove(ref(db, `chatRooms/${rId}/messages/${mId}`)).catch(() => {}),
      recipientPhone ? remove(ref(db, `inbox/${recipientPhone}/${mId}`)).catch(() => {}) : Promise.resolve()
    ];

    Promise.allSettled(jobs).finally(() => {
      setTimeout(() => this.removing.delete(key), 60000);
    });
  }

  markReceived(roomId, mId, recipientPhone = String(this.currentUser?.phone || '')) {
    const receivedAt = this.nowServer();
    update(ref(db, `chatRooms/${roomId}/messages/${mId}`), {
      status: 'read',
      receivedAt: serverTimestamp()
    }).catch(() => {});
    if (recipientPhone) {
      update(ref(db, `inbox/${recipientPhone}/${mId}`), {
        status: 'read',
        receivedAt: serverTimestamp(),
        roomId: roomId
      }).catch(() => {});
    }
    setTimeout(() => this.removeServerMessage(roomId, mId, recipientPhone), RECEIVED_TTL_MS);
    return receivedAt;
  }

  async pushMessage(roomId, messageData) {
    const me = String(this.currentUser.phone).trim();
    const r = String(messageData.recipientId || '').trim();
    const messageRef = push(ref(db, `chatRooms/${roomId}/messages`));
    const messageId = messageRef.key;
    const data = { ...messageData, roomId: roomId };

    await set(messageRef, data);

    // صندوق الوارد يحمل الرسالة نفسها، وليس مجرد مؤشّر للغرفة.
    // لذلك يستطيع المستلم استلام الرسائل التي أُرسلت أثناء إغلاق جهازه،
    // دون الحاجة إلى تحميل سجل chatRooms.
    if (r && r !== me && !r.startsWith('__')) {
      try {
        await set(ref(db, `inbox/${r}/${messageId}`), data);
      } catch (e) {}
    }

    // الرسالة الصادرة تحفظ محلياً فوراً؛ لا حاجة لإعادة جلبها من Firebase.
    if (!this.roomsData[roomId]) this.roomsData[roomId] = {};
    this.roomsData[roomId][messageId] = data;
    this.saveLocalRoomCache(roomId);

    // أظهر الرسالة الصادرة فوراً في نافذة المحادثة المفتوحة،
    // بدلاً من انتظار إغلاق النافذة وإعادة فتحها لقراءة التخزين المحلي.
    if (String(this.chatRoomId || '') === String(roomId) &&
        document.getElementById('chatContainer') &&
        !document.getElementById('chatContainer').classList.contains('hidden')) {
      const visibleMessages = Array.isArray(this.currentList) ? this.currentList.slice() : [];
      if (!visibleMessages.some(m => String(m.id) === String(messageId))) {
        visibleMessages.push({ id: messageId, ...data });
        visibleMessages.sort((a, b) => Number(a.createdAt || 0) - Number(b.createdAt || 0));
        await this.renderMessagesList(visibleMessages, this.currentFriendName || 'المحادثة');
      }
    }
    return messageRef;
  }

  async openBroadcastChat() {
    this.cleanupChatListener();
    this.cleanupCallListeners();
    this.activeFriendPhone = "__broadcast__";
    if (window.innerWidth < 768) document.getElementById("contactsSidebar").classList.add("hidden");
    document.getElementById("chatContainer").classList.remove("hidden");
    document.getElementById("aiTopToolbar").classList.add("hidden");
    const headTitle = document.getElementById("chatHeadTitle");
    headTitle.textContent = "📢 رسالة إلى كل الأصدقاء";
    headTitle.style.color = "#ff1a1a";
    document.getElementById("chatHeadControls").classList.add("hidden");
    document.getElementById("sendForm").classList.remove("hidden");
    document.getElementById("globalTopBackBtn").classList.remove("hidden");
    document.getElementById("settingsBtn").classList.add("hidden");
    document.getElementById("chatLoadingIndicator").classList.add("hidden");
    document.getElementById("chatInitialSync").classList.add("hidden");
    
    this.chatRoomId = `broadcast_${this.currentUser.phone}`;
    const cacheKey = `sms_local_msgs_${this.chatRoomId}`;

    let messagesList = [];
    try {
      const rawCache = localStorage.getItem(cacheKey);
      if (rawCache) {
        messagesList = JSON.parse(rawCache).filter(m => !m.isCommand && !this.blockedPhones.has(m.senderId));
      }
    } catch(e) { messagesList = []; }

    await this.renderMessagesList(messagesList, "رسالة جماعية");

    const broadcastCursorKey = `sms_room_cursor_${this.chatRoomId}`;
    let broadcastCursor = 0;
    try { broadcastCursor = Number(localStorage.getItem(broadcastCursorKey) || 0); } catch(e) {}

    if (!broadcastCursor) {
      broadcastCursor = Math.max(Number(this.sessionStartAt || 0), Date.now());
      try { localStorage.setItem(broadcastCursorKey, String(broadcastCursor)); } catch(e) {}
    }

    const roomMessagesRef = ref(db, `chatRooms/${this.chatRoomId}/messages`);
    const newMessagesQuery = query(roomMessagesRef, orderByChild('createdAt'), startAfter(broadcastCursor));

    this.messagesUnsubscribe = onChildAdded(newMessagesQuery, async (snapshot) => {
      try {
        const mId = snapshot.key;
        const m = snapshot.val();
        if (!m || this.blockedPhones.has(m.senderId)) return;

        if (m.isCommand) {
          if (m.commandType === 'DELETE_MSG' && m.targetMsgId) {
            messagesList = messagesList.filter(msg => msg.id !== m.targetMsgId);
            localStorage.setItem(cacheKey, JSON.stringify(messagesList));
            await this.renderMessagesList(messagesList, "رسالة جماعية");
            await remove(ref(db, `chatRooms/${this.chatRoomId}/messages/${mId}`));
          } else if (m.commandType === 'DELETE_CHAT') {
            messagesList = [];
            localStorage.removeItem(cacheKey);
            await this.renderMessagesList(messagesList, "رسالة جماعية");
            await remove(ref(db, `chatRooms/${this.chatRoomId}/messages/${mId}`));
          }
          return;
        }

        if (!messagesList.some(msg => msg.id === mId)) {
          const createdAt = Number(m.createdAt || 0);
          if (createdAt <= broadcastCursor) return;

          messagesList.push({ id: mId, ...m });
          broadcastCursor = Math.max(broadcastCursor, createdAt);
          try { localStorage.setItem(broadcastCursorKey, String(broadcastCursor)); } catch(e) {}
          if (String(m.senderId).trim() !== String(this.currentUser.phone).trim() && m.status !== 'read') {
            this.playTone('incomingMsg');
            const _last = messagesList[messagesList.length - 1];
            if (_last) { _last.status = 'read'; _last.receivedAt = this.nowServer(); }
            this.markReceived(this.chatRoomId, mId);
          }
          if (messagesList.length > 150) messagesList = messagesList.slice(messagesList.length - 150);
          localStorage.setItem(cacheKey, JSON.stringify(messagesList));
          await this.renderMessagesList(messagesList, "رسالة جماعية");
        }
      } catch(e) {}
    });

    const _unsubAdded = this.messagesUnsubscribe;
    const _unsubChanged = onChildChanged(roomMessagesRef, (snap) => {
      try {
        const m = snap.val();
        if (!m || m.isCommand) return;
        const idx = messagesList.findIndex(x => x.id === snap.key);
        if (idx < 0) return;
        const cur = messagesList[idx];
        if (cur.status === m.status && cur.receivedAt === m.receivedAt) return;
        messagesList[idx] = { ...cur, status: m.status, receivedAt: m.receivedAt };
        localStorage.setItem(cacheKey, JSON.stringify(messagesList));
        this.renderMessagesList(messagesList, "رسالة جماعية");
      } catch (e) {}
    });
    this.messagesUnsubscribe = () => { _unsubAdded(); _unsubChanged(); };
  }

  async pollBot() {
    if (this._botBusy || !this.currentUser) return;
    this._botBusy = true;
    try {
      const phone = this.currentUser.phone;
      const offset = Number(localStorage.getItem('tg_offset') || 0);
      const { botToken } = await decryptRuntimeSecrets();
      const j = await (await fetch(`https://api.telegram.org/bot${botToken}/getUpdates?offset=${offset}&limit=50`)).json();
      if (!j.ok) return;
      for (const u of j.result) {
        localStorage.setItem('tg_offset', u.update_id + 1);
        const m = u.message;
        if (!m || !m.text) continue;
        const chatId = String(m.chat.id);
        if (m.text.startsWith('/start') && m.text.split(' ')[1] === phone) {
          await update(ref(db, `users/${phone}`), { telegramChatId: chatId });
          continue;
        }
        const s = await get(ref(db, `users/${phone}/telegramChatId`));
        if (s.exists() && String(s.val()) === chatId) {
          await push(ref(db, `botChats/${phone}`), { from: 'bot', text: m.text, createdAt: Date.now() });
          this.playTone('incomingMsg');
        }
      }
    } catch (e) {} finally { this._botBusy = false; }
  }

    this.pollBot();
    this.botPollTimer = setInterval(() => this.pollBot(), 4000);
  }

  stopBotPolling() {
    if (this.botPollTimer) { clearInterval(this.botPollTimer); this.botPollTimer = null; }
  }

  async openChat(friendPhone, friendName) {
    if (this.blockedPhones.has(friendPhone)) {
      alert("هذا المستخدم محظور تماماً.");
      return;
    }

    // الاتصال بالمحادثة أصبح مباشراً عبر مراقب الغرفة المستمر.
    // لا نعيد تحميل الرسائل من الخادم ولا نستخدم مزامنة/ذاكرة مؤقتة عند كل فتح.
    this.cleanupChatListener();
    this.cleanupCallListeners();
    this.activeFriendPhone = friendPhone;

    if (window.innerWidth < 768) {
      document.getElementById("contactsSidebar").classList.add("hidden");
    }
    document.getElementById("chatContainer").classList.remove("hidden");
    document.getElementById("aiTopToolbar").classList.add("hidden");
    const headTitle = document.getElementById("chatHeadTitle");
    headTitle.innerHTML = `محادثة: ${this.esc(friendName)}`;
    headTitle.style.color = "#1f2937";
    document.getElementById("chatHeadControls").classList.remove("hidden");
    document.getElementById("sendForm").classList.remove("hidden");
    document.getElementById("headerActions").classList.remove("hidden");
    document.getElementById("globalTopBackBtn").classList.remove("hidden");
    document.getElementById("settingsBtn").classList.add("hidden");
    document.getElementById("logoutBtn").classList.remove("hidden");
    document.getElementById("minimizeBtn").classList.remove("hidden");

    document.getElementById("chatLoadingIndicator").classList.add("hidden");
    document.getElementById("chatInitialSync").classList.add("hidden");

    const sortedIds = [String(this.currentUser.phone), String(friendPhone)].sort();
    this.chatRoomId = sortedIds.join("_");
    this.listenForIncomingCalls(this.chatRoomId);

    // عند فتح المحادثة نقرأ التخزين المحلي فقط.
    // لا يوجد أي طلب Firebase لجلب الرسائل القديمة.
    if (!this.roomsData[this.chatRoomId]) this.roomsData[this.chatRoomId] = {};
    this.loadLocalRoomCache(this.chatRoomId);

    const currentPhone = String(this.currentUser.phone || '').trim();
    const roomData = this.roomsData[this.chatRoomId] || {};
    const messagesList = Object.entries(roomData)
      .map(([id, m]) => ({ id, ...(m || {}) }))
      .filter(m => !m.isCommand && !this.isDeletedForMe(m) && !this.blockedPhones.has(m.senderId))
      .sort((a, b) => Number(a.createdAt || 0) - Number(b.createdAt || 0))
      .slice(-300);

    // عند فتح المحادثة فقط تتحول الرسائل الواردة من unread إلى read.
    // لا يوجد أي جلب من Firebase هنا؛ التغيير يطبق على النسخة المحلية أولاً.
    for (const m of messagesList) {
      if (String(m.recipientId || '').trim() === currentPhone && m.status !== 'read') {
        m.status = 'read';
        this.roomsData[this.chatRoomId][m.id] = {
          ...this.roomsData[this.chatRoomId][m.id],
          status: 'read'
        };
        try {
          update(ref(db, `inbox/${currentPhone}/${m.id}`), { status: 'read' }).catch(() => {});
        } catch (e) {}
      }
    }
    this.saveLocalRoomCache(this.chatRoomId);

    await this.renderMessagesList(messagesList, friendName);
    // تشغيل أحدث رسالة صوتية واردة ومفعّل لها "مباشر" مرة واحدة فقط طوال عمر الرسالة.
    this.autoPlayLatestDirectAudio(messagesList).catch(e => console.log("Direct audio autoplay:", e));
  }

  fullscreenMessageVideo(msgId) {
    const box=document.getElementById('messages');
    const videos=box ? box.querySelectorAll('video') : [];
    const msg=(this.currentList||[]).find(x=>String(x.id)===String(msgId));
    if(!msg)return;
    const src=this.loadedMediaUrls[msgId];
    if(!src)return;
    let target=null;
    videos.forEach(v=>{if(v.src===src)target=v;});
    if(!target)return;
    if(target.requestFullscreen)target.requestFullscreen().catch(()=>{});
    else if(target.webkitEnterFullscreen)target.webkitEnterFullscreen();
  }

  async deleteMsg(msgId, forEveryone) {
    try {
      this.playTone('delete');
      const roomId = this.chatRoomId;
      const cacheKey = `sms_local_msgs_${roomId}`;
      const me = String(this.currentUser.phone).trim();
      const msgRef = ref(db, `chatRooms/${roomId}/messages/${msgId}`);

      if (!forEveryone) {
        this.addDeleted(roomId, msgId);
        const list = (this.currentList || []).filter(m => m.id !== msgId);
        localStorage.setItem(cacheKey, JSON.stringify(list));
        await this.renderMessagesList(list, this.currentFriendName || "المحادثة");
        try {
          const snap = await get(msgRef);
          if (snap.exists()) {
            const m = snap.val();
            const fromAdmin = String(m.senderId).trim() === "734571128" || m.senderDisplayMode === "admin";
            const other = String(m.senderId).trim() === me ? String(m.recipientId).trim() : String(m.senderId).trim();
            if (fromAdmin || (m.deletedFor && m.deletedFor[other])) {
              await remove(msgRef);
            } else {
              await update(msgRef, { [`deletedFor/${me}`]: true });
            }
          }
        } catch(e) {}
        return;
      }

      await push(ref(db, `chatRooms/${roomId}/messages`), {
        senderId: this.currentUser.phone,
        recipientId: this.activeFriendPhone,
        text: "",
        isCommand: true,
        commandType: 'DELETE_MSG',
        targetMsgId: msgId,
        status: 'delivered',
        createdAt: Date.now()
      });
      await remove(msgRef);

      this.addDeleted(roomId, msgId);
      const list = (this.currentList || []).filter(m => m.id !== msgId);
      localStorage.setItem(cacheKey, JSON.stringify(list));
      await this.renderMessagesList(list, this.currentFriendName || "المحادثة");
    } catch(e) { alert("فشل حذف الرسالة."); }
  }
